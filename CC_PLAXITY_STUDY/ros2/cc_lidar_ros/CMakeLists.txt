cmake_minimum_required(VERSION 3.8)
project(cc_lidar_ros CXX)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(lidar_ros src/lidar_ros.cpp)
target_compile_options(lidar_ros PRIVATE -O2)
ament_target_dependencies(lidar_ros rclcpp sensor_msgs std_msgs)
add_executable(lidar_ros_cl src/lidar_ros_cl.cpp)
target_compile_options(lidar_ros_cl PRIVATE -O2)
ament_target_dependencies(lidar_ros_cl rclcpp sensor_msgs std_msgs)
add_executable(lidar_ros_cl2 src/lidar_ros_cl2.cpp)
target_compile_options(lidar_ros_cl2 PRIVATE -O2)
ament_target_dependencies(lidar_ros_cl2 rclcpp sensor_msgs std_msgs)
add_executable(lidar_ros_cl3 src/lidar_ros_cl3.cpp)
target_compile_options(lidar_ros_cl3 PRIVATE -O2)
ament_target_dependencies(lidar_ros_cl3 rclcpp sensor_msgs std_msgs)
add_executable(lidar_ros_tm src/lidar_ros_tm.cpp)
target_compile_options(lidar_ros_tm PRIVATE -O2)
ament_target_dependencies(lidar_ros_tm rclcpp sensor_msgs std_msgs)
add_executable(vo_ros_tm src/vo_ros_tm.cpp)
target_compile_options(vo_ros_tm PRIVATE -O2)
ament_target_dependencies(vo_ros_tm rclcpp sensor_msgs std_msgs)
install(TARGETS lidar_ros lidar_ros_cl lidar_ros_cl2 lidar_ros_cl3 lidar_ros_tm vo_ros_tm DESTINATION lib/${PROJECT_NAME})
ament_package()
